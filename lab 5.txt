// Home Screen
@Composable
fun HomeScreen(
    onWebsiteClick: () -> Unit,
    onCallClick: () -> Unit,
    onCoursesClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    Column(
        modifier = modifier
            .fillMaxSize()
            .padding(24.dp),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {
        Text(
            "CSC Courses",
            style = MaterialTheme.typography.headlineMedium
        )

        Button(onClick = onWebsiteClick) {
            Text("University website")
        }

        Button(onClick = onCallClick) {
            Text("Call the department")
        }

        Button(onClick = onCoursesClick) {
            Text("Browse courses")
        }
    }
}

@Composable
fun HomeRoute(onCoursesClick: () -> Unit) {
    val context = LocalContext.current

    HomeScreen(
        onWebsiteClick = {
            val intent = Intent(
                Intent.ACTION_VIEW,
                Uri.parse("https://www.iau.edu.sa")
            )

            try {
                context.startActivity(intent)
            } catch (e: ActivityNotFoundException) {
                Toast.makeText(
                    context,
                    "No app can open this",
                    Toast.LENGTH_SHORT
                ).show()
            }
        },
        onCallClick = {
            val intent = Intent(
                Intent.ACTION_DIAL,
                Uri.parse("tel:0138000000")
            )

            try {
                context.startActivity(intent)
            } catch (e: ActivityNotFoundException) {
                Toast.makeText(
                    context,
                    "No app can open this",
                    Toast.LENGTH_SHORT
                ).show()
            }
        },
        onCoursesClick = onCoursesClick
    )
}

Task 3 — navigation/CourseApp.kt

package sa.edu.iau.cshj.lab05.navigation

import androidx.compose.runtime.Composable
import androidx.navigation.compose.NavHost
import androidx.navigation.compose.composable
import androidx.navigation.compose.rememberNavController
import androidx.navigation.toRoute
import sa.edu.iau.cshj.lab05.ui.HomeRoute
import sa.edu.iau.cshj.lab05.ui.CourseListRoute
import sa.edu.iau.cshj.lab05.ui.CourseDetailRoute

@Composable
fun CourseApp() {
    val navController = rememberNavController()

    NavHost(
        navController = navController,
        startDestination = Home
    ) {
        composable<Home> {
            HomeRoute(
                onCoursesClick = {
                    navController.navigate(CourseList)
                }
            )
        }

        composable<CourseList> {
            CourseListRoute(
                onCourseClick = { code ->
                    navController.navigate(
                        CourseDetail(code)
                    )
                }
            )
        }

        composable<CourseDetail> { backStackEntry ->
            val detail: CourseDetail =
                backStackEntry.toRoute()

            CourseDetailRoute(
                code = detail.code,
                onBackClick = {
                    navController.popBackStack()
                }
            )
        }
    }
}

@Composable
fun CourseDetailRoute(
    code: String,
    onBackClick: () -> Unit
) {
    val context = LocalContext.current

    CourseDetailScreen(
        uiState = CourseDetailUiState(code = code),
        onShareClick = {
            val send = Intent(Intent.ACTION_SEND).apply {
                putExtra(
                    Intent.EXTRA_TEXT,
                    "I'm taking $code this semester!"
                )
                type = "text/plain"
            }

            context.startActivity(
                Intent.createChooser(send, "Share via")
            )
        },
        onBackClick = onBackClick
    )
}
