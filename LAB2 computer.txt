fun finalMark(
    quizzes: Double,
    project: Double,
    finalExam: Double
): Double = quizzes * 0.10 + project * 0.70 + finalExam * 0.20


fun printBanner(title: String, width: Int = 40) {
    println("-".repeat(width))
    println(title)
    println("-".repeat(width))
}

fun letterGrade(mark: Int): String = when {
    mark >= 95 && mark <= 100 -> "A+"
    mark >= 90 && mark <= 94 -> "A"
    mark >= 85 && mark <= 89 -> "B+"
    mark >= 80 && mark <= 84 -> "B"
    mark >= 75 && mark <= 79 -> "C+"
    mark >= 70 && mark <= 74 -> "C"
    mark >= 65 && mark <= 69 -> "D+"
    mark >= 60 && mark <= 64 -> "D"
    mark >= 0 && mark <= 59 -> "F"
    else -> "Invalid"
}
fun status(mark: Int): String =
    if (mark >= 60 && mark <= 100) "Pass" else "Fail"

fun describeTeam(teamName: String?): String {
    return teamName?.let {
        "Team: $it (${it.length} characters)"
    } ?: "Team not registered yet"
}

data class Course(
    val code: String,
    val title: String,
    val credits: Int,
    val days: String
)

fun findCourse(list: List<Course>, code: String): String {
    return list.firstOrNull { it.code == code }?.title ?: "Course not found"
}
